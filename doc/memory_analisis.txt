============================================= MEMORY ANALISIS =================================


La Game Boy usa un BUS de direcciones de 16-bits es decir el tipo de dato que recibe
es obligatoriamente un std::uint16_t un tipo de datos de 16bits


De momento no incluiremos los MBC, probaremos con ROMS pequeñas como tetris que 
caben dentro de los 32kb de ram.


Clase MemoryBus {
    Privado:
        bytes wram[8192] 8 KIB ram 
	bytes vram[8192] 8 KIb vram
	bytes oam[160] 160 bytes para almacenar datos del objeto 
	bytes hram[127] 127bytes del silicio	

	bytes interrupt_enable; // 1 byte en 0xFFFF

	bytes vector cartridge_rom; // Arreglo de 32kb para el cartucho del juego
	puntero_inteligente cartridge_mbc; // nos indica que MBC usaremos dependiendo del juego

    Público:
	
	Metodo load_rom(vector de 8 bits con los datos de la rom) // carga el juego en memoria del bus
        Método read(dirección_16bits) -> devuelve byte {
            devolver memoria_plana[dirección_16bits]
        }

        Método write(dirección_16bits, valor_byte) {
            memoria_plana[dirección_16bits] = valor_byte
        }
}




=============== Clase Interfaz IMBC =========================
// IMBC.h - Interfaz independiente para cualquier controlador de banco
#ifndef I_MBC_H
#define I_MBC_H

#include <cstdint>

class IMBC {
public:
    virtual ~IMBC() = default;
    virtual std::uint8_t read_rom(std::uint16_t address) = 0;
    virtual void write_rom(std::uint16_t address, std::uint8_t value) = 0;
    virtual std::uint8_t read_ram(std::uint16_t address) = 0;
    virtual void write_ram(std::uint16_t address, std::uint8_t value) = 0;
};

#endif
